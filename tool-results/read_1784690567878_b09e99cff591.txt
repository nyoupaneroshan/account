   164→   ANIMATED COUNTER HOOK (UNCHANGED)
   165→   ──────────────────────────────────────────── */
   166→
   167→function useAnimatedCounter(target: number, duration = 2000, startOnView = true) {
   168→  const [count, setCount] = useState(0)
   169→  const [hasStarted, setHasStarted] = useState(false)
   170→  const ref = useRef<HTMLDivElement>(null)
   171→
   172→  const startCounting = useCallback(() => {
   173→    if (hasStarted) return
   174→    setHasStarted(true)
   175→    const startTime = performance.now()
   176→    const isDecimal = target % 1 !== 0
   177→
   178→    const animate = (currentTime: number) => {
   179→      const elapsed = currentTime - startTime
   180→      const progress = Math.min(elapsed / duration, 1)
   181→      // Ease-out cubic
   182→      const eased = 1 - Math.pow(1 - progress, 3)
   183→      const current = eased * target
   184→      setCount(isDecimal ? parseFloat(current.toFixed(1)) : Math.floor(current))
   185→      if (progress < 1) {
   186→        requestAnimationFrame(animate)
   187→      } else {
   188→        setCount(target)
   189→      }
   190→    }
   191→    requestAnimationFrame(animate)
   192→  }, [target, duration, hasStarted])
   193→
   194→  useEffect(() => {
   195→    if (!startOnView || !ref.current) return
   196→    const observer = new IntersectionObserver(
   197→      ([entry]) => {
   198→        if (entry.isIntersecting) {
   199→          startCounting()
   200→        }
   201→      },
   202→      { threshold: 0.3 }
   203→    )
   204→    observer.observe(ref.current)
   205→    return () => observer.disconnect()
   206→  }, [startCounting, startOnView])
   207→
   208→  return { count, ref }
   209→}
