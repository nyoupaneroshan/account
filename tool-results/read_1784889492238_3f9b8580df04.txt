  1199→/**
  1200→ * Get a single translation value by key and locale
  1201→ */
  1202→export function t(key: keyof TranslationStrings, locale: Locale = 'en'): string {
  1203→  const trans = getTranslations(locale)
  1204→  return trans[key] || translations.en[key] || key
  1205→}

  1207→/**

  1211→export function useTranslation(locale: Locale = 'en') {
  1212→  const trans = getTranslations(locale)
  1213→
  1214→  return {
