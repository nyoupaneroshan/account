    38→/* ────────────────────────────────────────────
    39→   PASSWORD STRENGTH HELPER
    40→   ──────────────────────────────────────────── */
    41→function getPasswordStrength(password: string): {
    42→  score: number; label: string; color: string
    43→} {
    44→  if (!password) return { score: 0, label: '', color: '' }
    45→  let score = 0
    46→  if (password.length >= 6) score++
    47→  if (password.length >= 10) score++
    48→  if (/[A-Z]/.test(password)) score++
    49→  if (/[0-9]/.test(password)) score++
    50→  if (/[^A-Za-z0-9]/.test(password)) score++
    51→
    52→  if (score <= 1) return { score: 1, label: 'Weak', color: '#ef4444' }
    53→  if (score <= 2) return { score: 2, label: 'Fair', color: '#f97316' }
    54→  if (score <= 3) return { score: 3, label: 'Good', color: '#eab308' }
    55→  if (score <= 4) return { score: 4, label: 'Strong', color: '#22c55e' }
    56→  return { score: 5, label: 'Very Strong', color: '#10B981' }
    57→}
