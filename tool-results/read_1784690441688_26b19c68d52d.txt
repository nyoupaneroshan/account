    20→/* ────────────────────────────────────────────
    21→   PASSWORD STRENGTH HELPER
    22→   ──────────────────────────────────────────── */
    23→function getPasswordStrength(password: string): {
    24→  score: number; label: string; color: string
    25→} {
    26→  if (!password) return { score: 0, label: '', color: '' }
    27→  let score = 0
    28→  if (password.length >= 6) score++
    29→  if (password.length >= 10) score++
    30→  if (/[A-Z]/.test(password)) score++
    31→  if (/[0-9]/.test(password)) score++
    32→  if (/[^A-Za-z0-9]/.test(password)) score++
    33→
    34→  if (score <= 1) return { score: 1, label: 'Weak', color: '#ef4444' }
    35→  if (score <= 2) return { score: 2, label: 'Fair', color: '#f97316' }
    36→  if (score <= 3) return { score: 3, label: 'Good', color: '#eab308' }
    37→  if (score <= 4) return { score: 4, label: 'Strong', color: '#22c55e' }
    38→  return { score: 5, label: 'Very Strong', color: '#10B981' }
    39→}

    44→function AnimatedInput({
    45→  id,
    46→  label,
    47→  type = 'text',
    48→  placeholder,
    49→  value,
    50→  onChange,
    51→  onKeyDown,
    52→  icon: Icon,
    53→  rightElement,
    54→  className,
    55→  hint,
    56→}: {
    57→  id: string
    58→  label: string
    59→  type?: string
    60→  placeholder: string
    61→  value: string
    62→  onChange: (e: React.ChangeEvent<HTMLInputElement>) => void
    63→  onKeyDown?: (e: React.KeyboardEvent<HTMLInputElement>) => void
    64→  icon?: React.ElementType
    65→  rightElement?: React.ReactNode
    66→  className?: string
    67→  hint?: string
    68→}) {
    69→  const [focused, setFocused] = useState(false)
    70→  const hasValue = value.length > 0
    71→
    72→  return (
    73→    <div className="space-y-1.5">
    74→      <Label htmlFor={id} className={cn(
    75→        "text-sm font-medium transition-colors duration-200",
    76→        focused ? "text-emerald-400" : "text-zinc-300"
    77→      )}>
    78→        {label}
    79→      </Label>
    80→      <div className={cn(
    81→        "relative rounded-lg transition-all duration-200",
    82→        focused && "ring-1 ring-emerald-500/30"
    83→      )}>
    84→        {Icon && (
    85→          <div className={cn(
    86→            "absolute left-3 top-1/2 -translate-y-1/2 transition-colors duration-200",
    87→            focused ? "text-emerald-400" : "text-zinc-600"
    88→          )}>
    89→            <Icon className="h-4 w-4" />
    90→          </div>
    91→        )}
    92→        <Input
    93→          id={id}
    94→          type={type}
    95→          placeholder={placeholder}
    96→          value={value}
    97→          onChange={onChange}
    98→          onKeyDown={onKeyDown}
    99→          onFocus={() => setFocused(true)}
   100→          onBlur={() => setFocused(false)}
   101→          className={cn(
   102→            "bg-white/[0.04] border-white/[0.08] text-white placeholder:text-zinc-600",
   103→            "focus-visible:ring-0 focus-visible:border-emerald-500/50",
   104→            "transition-all duration-200",
   105→            Icon && "pl-10",
   106→            rightElement && "pr-10",
   107→            className
   108→          )}
   109→        />
   110→        {rightElement && (
   111→          <div className="absolute right-3 top-1/2 -translate-y-1/2">
   112→            {rightElement}
   113→          </div>
   114→        )}
   115→      </div>
   116→      {hint && (
   117→        <p className={cn(
   118→          "text-[11px] transition-all duration-300 overflow-hidden",
   119→          focused || hasValue ? "max-h-6 opacity-100" : "max-h-0 opacity-0"
   120→        )}>
   121→          <span className="text-zinc-600">{hint}</span>
   122→        </p>
   123→      )}
   124→    </div>
   125→  )
   126→}
