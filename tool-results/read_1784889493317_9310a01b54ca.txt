    74→export function SettingsView() {
    75→  const {
    76→    mode, setMode, currentOrgId, currentFiscalYear,
    77→    language, setLanguage, currentUser,
    78→  } = useAppStore()
    79→  const [saving, setSaving] = useState(false)
    80→  const [loading, setLoading] = useState(true)
    81→  const [theme, setTheme] = useState('light')
    82→  const initialLoadDone = useRef(false)
    83→
    84→  // Org settings form
    85→  const [orgForm, setOrgForm] = useState({
    86→    name: '',
    87→    nameNepali: '',
    88→    pan: '',
    89→    address: '',
    90→    city: '',
    91→    province: '',
    92→    phone: '',
    93→    email: '',
    94→  })
    95→
    96→  // Tax configuration
    97→  const [taxConfig, setTaxConfig] = useState({
    98→    vatEnabled: true,
    99→    vatRate: `${Math.round(NEPAL_VAT_RATE * 100)}`,

   104→  })
   105→

   107→  const [invoiceConfig, setInvoiceConfig] = useState({
   108→    prefix: 'INV-',
   109→    nextNumber: '001',
   110→    defaultTerms: 'Payment due within 30 days.',
   111→  })
   112→
   113→  // Preferences
   114→  const [preferences, setPreferences] = useState({
   115→    language: language,
   116→    theme: 'light',
   117→    defaultMode: mode,
   118→  })
   119→
   120→  // Load org data from proper API
   121→  useEffect(() => {
   122→    if (!currentOrgId) return

   124→      setLoading(true)
   125→      try {
   126→        const res = await authFetch(`/api/settings?orgId=${currentOrgId}`)
   127→        if (res.ok) {
   128→          const data = await res.json()

   130→            name: data.name || '',
   131→            nameNepali: data.nameNepali || '',
   132→            pan: data.panNumber || '',

   135→            province: data.province || '',
   136→            phone: data.phone || '',
   137→            email: data.email || '',

   140→            ...prev,
   141→            vatEnabled: data.vatEnabled ?? true,

   175→        method: 'PUT',
   176→        headers: { 'Content-Type': 'application/json' },
   177→        body: JSON.stringify({ language: preferences.language }),
   178→      }).catch(() => {
   179→        // silently fail - language is already saved in org settings and Zustand
   180→      })

   193→      const res = await authFetch('/api/settings', {
   194→        method: 'PUT',
   195→        headers: { 'Content-Type': 'application/json' },

   197→          id: currentOrgId,
   198→          name: orgForm.name,
   199→          nameNepali: orgForm.nameNepali || null,
   200→          panNumber: orgForm.pan || null,
   201→          address: orgForm.address || null,
   202→          city: orgForm.city || null,
   203→          province: orgForm.province || null,
   204→          phone: orgForm.phone || null,
   205→          email: orgForm.email || null,
   206→          vatEnabled: taxConfig.vatEnabled,
   207→          vatRate: taxConfig.vatRate,
   208→          tdsEnabled: taxConfig.tdsEnabled,
   209→          ssfEnabled: taxConfig.ssfEnabled,
   210→          invoicePrefix: invoiceConfig.prefix,
   211→          invoiceNextNumber: invoiceConfig.nextNumber,
   212→          invoiceDefaultTerms: invoiceConfig.defaultTerms,
   213→          mode: preferences.defaultMode,
   214→          language: preferences.language,
   215→        }),

   217→      const data = await res.json()

   219→        if (showToast) toast.success('Settings saved successfully')

   221→        if (showToast) toast.error(data.error || 'Failed to save settings')
   222→      }
   223→    } catch {
   224→      if (showToast) toast.error('Failed to save settings')
   225→    } finally {
   226→      setSaving(false)
   227→    }
   228→  }
   229→
   230→  // Auto-save settings when values change (debounced, silent)
   231→  // Only auto-save AFTER the initial load is complete to avoid saving data we just fetched
   232→  useEffect(() => {
   233→    if (!currentOrgId || loading || !initialLoadDone.current) return

   235→      handleSaveOrg(false)
   236→    }, 1500)
   237→    return () => clearTimeout(timer)
   238→  }, [orgForm, taxConfig, invoiceConfig, preferences])
   239→
   240→  const handleDeleteOrg = async () => {
   241→    toast.error('Organization deletion is disabled for safety. Contact support.')
   242→  }
   243→
   244→  if (loading) {
   245→    return (
   246→      <div className="p-4 md:p-6 space-y-6 max-w-4xl">
   247→        <div className="h-8 w-64 bg-muted/50 rounded animate-pulse" />
   248→        <div className="h-64 bg-muted/30 rounded-lg animate-pulse" />
   249→        <div className="h-48 bg-muted/30 rounded-lg animate-pulse" />
   250→      </div>
   251→    )
   252→  }

   259→          <h1 className="text-2xl font-bold tracking-tight flex items-center gap-2">
   260→            <Settings className="h-6 w-6" />
   261→            {t('settings', language as 'en' | 'ne' | 'hi')}
   262→          </h1>
   263→          <p className="text-sm text-muted-foreground">सेटिङ — Configure your accounting preferences</p>
   264→        </div>
   265→        <Button onClick={handleSaveOrg} disabled={saving}>
   266→          {saving ? <Loader2 className="h-4 w-4 mr-2 animate-spin" /> : <Save className="h-4 w-4 mr-2" />}
   267→          {t('save', language as 'en' | 'ne' | 'hi')}

   285→                id="orgName"

   289→              />
   290→            </div>

   299→            </div>
